\tdplotsetmaincoords{75}{25}
\begin{tikzpicture}[
	point/.style={circle,draw,very thin,fill,inner sep=0pt,minimum size=4pt},
	vector/.style={-latex},
	tdplot_main_coords
]

	% plane through the origin with normal vector (1,-1,2)
	% orthog basis used to draw sides: (1,1,0) and (1,-1,-1)

	% plane corners
	% sides of plane are vectors (3.5,3.5,0) and (2.25,-2.25,-2.25)
	\coordinate (P1) at (-1.5,-0.5,0.5);
	\coordinate (P2) at (2,3,0.5);
	\coordinate (P3) at (4.25,0.75,-1.75);
	\coordinate (P4) at (0.75,-2.75,-1.75);
	% plane vectors
	\coordinate (u) at (1.5,2,0.25);
	\coordinate (v) at (3,0,-1.5);
	% origin
	\coordinate (o) at (0,0,0);
	% x-axis
	\coordinate (x_begin) at (-2,0,0);
	\coordinate (x_trans) at (-1.75,0,0);
	\coordinate (x_end) at (4.5,0,0);
	% y-axis
	\coordinate (y_begin) at (0,-3.5,0);
	\coordinate (y_trans) at (0,3.5,0);
	\coordinate (y_end) at (0,5,0);
	% z-axis
	\coordinate (z_begin) at (0,0,-2.25);
	\coordinate (z_end) at (0,0,1.5);

	% fill/border for plane
	\filldraw[opacity=0.25,gray] (P1) to (P2) to (P3) to (P4) to cycle;
	\draw[dotted]  (P1) to (P2) to (P3) to (P4) to cycle;

	% x-axis
	\draw (x_begin) to (x_trans);
	\draw[dashed] (x_trans) to (o);
	\draw[->] (o) to (x_end) node[right]{$x$};

	% y-axis
	\draw (y_begin) to (o);
	\draw[dashed] (o) to (y_trans);
	\draw[->] (y_trans) to (y_end) node[above] {$y$};

	% z-axis
	\draw[dashed] (z_begin) to (o);
	\draw[->] (o) to (z_end) node[right] {$z$};

	% vectors
	\draw[vector,thick] (0,0,0) to node[right,at end] {$\uvec{u}$} (u);
	\draw[vector,thick] (0,0,0) to node[below left,near end] {$\uvec{v}$} (v);
	\draw[densely dotted] (u) to (v);

	\tdplotdefinepoints(0,0,0)(0.597,0.796,0.1)(0.895,0,-0.447)
	\tdplotdrawpolytopearc[]{1}{anchor=south west}{$\theta$}

\end{tikzpicture}
