\tdplotsetmaincoords{100}{30}
\begin{tikzpicture}[
	point/.style={circle,draw,very thin,fill,inner sep=0pt,minimum size=4pt},
	vector/.style={-latex},
	tdplot_main_coords
]

	% plane through the origin with normal vector (1,-1,2)
	% orthog basis used to draw sides: (1,1,0) and (1,-1,-1)

	% origin
	\node[point] at (0,0,0) (o) [label=left:{$\zerovec$}] {};
	% plane corners
	% sides of plane are vectors (6,6,0) and (2.5,-2.5,-2.5)
	\coordinate (P1) at (-1.5,-0.5,0.5);
	\coordinate (P2) at (4.5,5.5,0.5);
	\coordinate (P3) at (7,3,-2);
	\coordinate (P4) at (1,-3,-2);
	% parallel vector
	\coordinate (u) at (5.25,3,-1.125);
	% external vector
	\coordinate (v) at (2,2,1.5);

	% fill/border for plane
	\filldraw[opacity=0.25,gray] (P1) to (P2) to (P3) to (P4) to cycle;
	\draw[dotted] (P1) to (P2) to (P3) to (P4) to cycle;

	% plane vectors
	\draw[vector] (o) to node[above,at end] {$\uvec{p}_1$} (2.25,3,0.375);
	\draw[vector] (o) to node[below left,near end] {$\uvec{p}_2$} (3,0,-1.5);
	\draw[vector] (o) to node[sloped,above,near end,xshift=0.5cm]
		{$\uvec{u}$ {\tiny (linearly dependent with $\uvec{p}_1,\uvec{p}_2$)}}
		(u);

	% indep vector
	\draw[vector] (o) to node[right,at end,yshift=-0.1cm] {$\uvec{v}$ {\tiny (linearly independent from $\uvec{p}_1,\uvec{p}_2$)}} (v);

\end{tikzpicture}
