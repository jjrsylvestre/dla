\tdplotsetmaincoords{95}{30}
\begin{tikzpicture}[
	point/.style={circle,draw,very thin,fill,inner sep=0pt,minimum size=4pt},
	vector/.style={-latex},
	tdplot_main_coords
]

	% plane through the origin with normal vector n = (1,-1,2)
	% orthog basis used to draw sides: (1,1,0) and (1,-1,-1)

	% plane corners
	% sides of plane are vectors (5.5,5.5,0) and (3,-3,-3)
	\coordinate (P1) at (-2,0,1);
	\coordinate (P2) at (3.5,5.5,1);
	\coordinate (P3) at (6.5,2.5,-2);
	\coordinate (P4) at (1,-3,-2);
	% normal vector terminal point
	\coordinate (n) at (1,-1,2);
	% smul of n to magic height: 1.5*n
	\coordinate (n') at (1.5,-1.5,3);
	% plane base point
	\node[point] at (0,0,0) (r) [label=left:$R$] {};
	% plane variable point
	\node[point] at (4,2,-1) (p) {};
	% external point: 1.5*n above p
	\node[point] at (5.5,0.5,2) (q) [label=right:$Q$] {};

	% fill/border for plane
	\filldraw[opacity=0.25,gray] (P1) to (P2) to (P3) to (P4) to cycle;
	\draw[dotted] (P1) to (P2) to (P3) to (P4) to cycle;
	\node[xshift=-0.5cm,yshift=0.25cm] at (P3) {$\Pi$};

	% normal vectors above r
	\draw[vector] (r) to (n');
	\draw[vector,very thick] (r) to node[left,near end] {$\uvec{n}$} (n);

	% base to external
	\draw[vector,very thick] (r) to node[above] {$\uvec{u}$} (q);

	% suggestive projection arrow:
	% 85% of the way up u and n'
	% but slightly shifted via 0.05*p
	\draw[->,densely dotted,thick] (4.475,0.325,1.75) to (1.475,-1.175,2.5);

	% rectangle sides
	\draw[dashed] (r) to (p) to node[right] {$d$} (q) to (n');

	% perp symbol @ p
	\draw (4.2,1.8,-0.6) to (3.8,1.6,-0.5) to (3.6,1.8,-0.9);


\end{tikzpicture}
