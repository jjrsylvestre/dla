\tdplotsetmaincoords{105}{30}
\begin{tikzpicture}[
	point/.style={circle,draw,very thin,fill,inner sep=0pt,minimum size=4pt},
	vector/.style={-latex},
	tdplot_main_coords
]

	% plane through the origin with normal vector (1,-1,2)
	% orthog basis used to draw sides: (1,1,0) and (1,-1,-1)

	% origin
	\node[point] at (0,0,0) (o) [label=left:{$\zerovec$}] {};
	% plane corners
	% sides of plane are vectors (4.5,4.5,0) and (3,-3,-3)
	\coordinate (P1) at (-1.5,0.5,1);
	\coordinate (P2) at (3,5,1);
	\coordinate (P3) at (6,2,-2);
	\coordinate (P4) at (1.5,-2.5,-2);
	% external vector
	\node[point] at (5.2,1.4,1.1) (v) {};
	% projection
	\node[point] at (4.2,2.4,-0.9) (p) {};
	% another point in the plane
	\node[point] at (4,1,-1.5) (u) {};

	% fill/border/label for plane
	\filldraw[opacity=0.25,gray] (P1) to (P2) to (P3) to (P4) to cycle;
	\draw[dotted]  (P1) to (P2) to (P3) to (P4) to cycle;
	\node[xshift=-0.5cm,yshift=0.25cm] at (P3) {$U$};

	% plane vectors
	\draw[vector,thick] (o) to node[below,sloped] {$\proj_U \uvec{v}$} (p);
	\draw[vector,thick] (o) to node[above] {$\uvec{v}$} (v);
	\draw[thick,dashed,blue] (p) to node[right,near end,sloped,rotate=-90] {$\min \dist$} (v);
	\draw[thick,dashed,red] (u) to (v);
	\draw[thin,dashed] (u) to (p);

	% perp symbol
	\draw[thin] (3.99,2.28,-0.855) to (4.09,2.18,-0.655) to (4.3,2.3,-0.7);

\end{tikzpicture}
