\tdplotsetmaincoords{105}{30}
\begin{tikzpicture}[
	point/.style={circle,draw,very thin,fill,inner sep=0pt,minimum size=4pt},
	vector/.style={-latex},
	tdplot_main_coords
]

	% origin
	\coordinate (o) at (0,0,0);
	% basis vectors for plane
	\coordinate (e1) at (1,1,0);
	\coordinate (e2) at (1,-1,-1);
	% but let's make the visualized e_2 a little longer
	\coordinate (e2_vis) at (1.5,-1.5,-1.5);
	% plane corners
	% sides of plane are vectors (4,4,0) and (3,-3,-3)
	\coordinate (p1) at (-2,0,1);
	\coordinate (p2) at (2,4,1);
	\coordinate (p3) at (5,1,-2);
	\coordinate (p4) at (1,-3,-2);
	% normal vector for plane
	\coordinate (n) at (1,-1,2);
	% perp space line endpoints
	%   (-3/4) n
	\coordinate (neg_n) at (-0.75,0.75,-1.5);
	%   (-1/2) n
	\coordinate (neg_n_plane) at (-0.5,0.5,-1);
	% external vector
	% v = (3/2) e_1 + (3/2) e_2 + (4/5) n
	% --> parallel to (38, -8, 1)
	\coordinate (v) at (3.8,-0.8,0.1);
	% proj v onto plane, equal to (3/2) e_1 + (3/2) e_2
	\coordinate (v_plane) at (3,0,-3/2);
	% proj v onto normal vector, equal to (4/5) n
	\coordinate (v_perp) at (0.8,-0.8,1.6);
	% end points for creating line spanned by v
	%   (5/4) * v
	\coordinate (E2) at (4.75,-1,0.125);
	%   (-5/8) * v
	\coordinate (neg_v) at (-2.375,0.5,-0.0625);
	%   (-3/8) * v (visually near plane boundary)
	\coordinate (neg_v_2) at (-1.425, 0.3, -0.0375);

	% fill/border/label for plane
	\filldraw[opacity=0.25,gray] (p1) to (p2) to (p3) to (p4) to cycle;
	\draw[dotted]  (p1) to (p2) to (p3) to (p4) to cycle;
	\node[xshift=-0.5cm,yshift=0.25cm] at (p3) {$E_{\lambda_1}$};
	% E_{\lambda_1} = \Span \{\uvec{e}_1, \uvec{e}_2\}

	% origin
	\node[point] at (o) [label={[yshift=0.1cm]left:{$\zerovec$}}] {};

	% perp space
	\draw (o) to (n) node[right] {$\orthogcmp{E}_{\lambda_1}$};
	\draw[dashed] (o) -- (neg_n_plane);
	\draw (neg_n) -- (neg_n_plane);

	% plane vectors
	\draw[vector,thick] (o) to node[above right,at end] {$\uvec{e}_1$} (e1);
	\draw[vector,thick] (o) to node[right,at end] {$\uvec{e}_2$} (e2_vis);
	% perp symbol: 20% in each direction
	\draw[thin] (0.2,0.2,0) to (0.4,0,-0.2) to (0.2,-0.2,-0.2);
	\draw[vector,thick] (o) to node[left,near end] {$\uvec{e}_3$} (v_perp);
	% \uvec{e}_3 = \proj_{\orthogcmp{E}_{\lambda_1}} \uvec{v} ;

	% other eigenspace
	\draw (neg_v) to (neg_v_2);
	\draw[dashed] (neg_v_2) to (o);
	\draw (o) to (E2) node[above right] {$E_{\lambda_2}$};
	% E_{\lambda_2} = \Span \{\uvec{v}\}
	\draw[vector,thick] (o) to (v) node[above] {$\uvec{v}$};
	\draw[dashed] (v_perp) to (v) to (v_plane) to (0.45,0,-0.225); % don't draw right to (o) because of e1/e2 perp symbol
	% perp symbol for projection of v onto plane:
	% (1/8) of v_perp, 10% of -(v_plane)
	\draw[thin] (3.1, -0.1, -1.3) to (2.8,-0.1,-1.15) to (2.7,0,-1.35);

\end{tikzpicture}
