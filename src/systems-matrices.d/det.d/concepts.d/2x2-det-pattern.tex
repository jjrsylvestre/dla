% general 2x2 det pattern
\begin{tikzpicture}[
	every left delimiter/.style={xshift=.5em,yscale=0.75,yshift=0.4em},
	every right delimiter/.style={xshift=-.5em,yscale=0.75,yshift=0.4em}
]
	\matrix (A)[
		matrix of math nodes,
		left delimiter={\lvert},
		right delimiter={\rvert}
	]{ a \amp b \\ c \amp d \\ };
	\node[left=-0.25em of A] {$\det A =$};
	\node[right=-0.25em of A] {$= a d - b c$.};
	\node[above left=-0.5em of A] {$+$};
	\node[above right=-0.5em of A] {$-$};
	\draw[-latex] (A.north west) to (A.south east);
	\draw[-latex] (A.north east) to (A.south west);
\end{tikzpicture}
