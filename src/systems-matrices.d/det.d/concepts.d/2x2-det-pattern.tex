\begin{tikzpicture}[
	every left delimiter/.style={xshift=.5em,yscale=0.75,yshift=0.4em},
	every right delimiter/.style={xshift=-.5em,yscale=0.75,yshift=0.4em},
	crisscross/.style={black!65!white}
]

	% want to artificially inflate width because
	% this image is part of a sidebyside with a p
	% and want more horizontal room for caption
	%
	% actual image dimensions are from (-0.8,-0.6) to (0.8,0.85)
	%
	\draw[black!5!white,very thin] (-1.4,-0.55) to (-1.4,-0.6) to (-1.35,-0.6);
	\draw[black!5!white,very thin] (1.4,0.85) to (1.4,0.85) to (1.35,0.85);

	\matrix (A)[
		matrix of math nodes,
		left delimiter={\lvert},
		right delimiter={\rvert}
	]{ a \amp b \\ c \amp d \\ };
	\node[crisscross,above left=-0.5em of A] {$+$};
	\node[crisscross,above right=-0.5em of A] {$-$};
	\begin{scope}[-latex,on background layer]
		\draw[crisscross] (A.north west) to (A.south east);
		\draw[crisscross] (A.north east) to (A.south west);
	\end{scope}
\end{tikzpicture}
